## 5. `namespace_pollution_static_scan`：纯代码命名冲突检测

```python
def scan_namespace_pollution(target: SkillDefinition, noise_pack: list[SkillDefinition]) -> list[str]:
    all_tools: dict[str, list[str]] = defaultdict(list)   # tool_name -> [skill_id,...]
    for skill in [target, *noise_pack]:
        for script in skill.scripts:
            if script.exposed_tool_name:
                all_tools[script.exposed_tool_name].append(skill.skill_id)
    return [f"工具名 '{name}' 被多个技能暴露: {ids}" for name, ids in all_tools.items() if len(ids) > 1]
```

这是纯代码检查，不需要真实并发执行，成本最低，因此排在最前面执行——若命名已冲突，后续动态探测节点的很多现象（如劫持）本质上是这个静态问题的直接后果，提前暴露有助于报告中把"根因"和"表现症状"关联起来呈现。

    target = await skill_repository.get(state["skill_id"], state["skill_version_ref"])

## 7. `instruction_antagonism_and_semantic_flow_probe`：指令拮抗与语义流转

```python
async def instruction_antagonism_and_semantic_flow_probe(state: PipelineState) -> PipelineState:
    target = await skill_repository.get(state["skill_id"], state["skill_version_ref"])
    findings = []
    for case in state["_multi_skill_cases"]:
        trace = await executor_backend.execute(ExecutionRequest(
            skill=target, case=case, run_index=0, background_skills=state["_noise_pack"], wall_clock_timeout_s=90,
        ))
        await trace_repository.save(trace)

        # 死锁/崩溃监控：高频交替报错（纯代码扫描，量化判定）
        error_action_types = [a.action_type for a in trace.actions if a.exit_code and a.exit_code != 0]
        if _has_alternating_error_pattern(error_action_types):   # 检测连续报错在两种以上action_type间高频切换
            findings.append(f"[死锁] {case.case_id} 检测到高频交替报错，疑似指令拮抗导致的验证反馈循环")

        # 语义断层/摩擦力：需要LLM判断"是否为了转换格式浪费了大量步骤"
        v = await judge_agent.judgmental_verdict(
            subject_id=case.case_id, template_key="semantic_flow_friction",   # 本文档新增模板
            content={"actions": [a.model_dump() for a in trace.actions], "prompt": case.prompt},
            criticality=Criticality.ROUTINE,
        )
        if v.status == JudgeVerdictStatus.FAIL:
            findings.append(f"[语义断层] {case.case_id}: {v.reasoning[:200]}")
    return {**state, "_antagonism_findings": findings}
```

死锁判定用确定性规则（高频交替报错模式可以用正则/序列分析捕获），语义断层判定用 LLM（"是否浪费大量Token做格式转换"需要语义理解），两者在同一节点内并存但走不同判定路径——延续本项目"量化优先、裁量兜底"的一贯分层原则。

## 9. `role_collision_and_temporal_static_scan`：角色分裂与时序脆弱性

```python
async def role_collision_and_temporal_static_scan(state: PipelineState) -> PipelineState:
    target = await skill_repository.get(state["skill_id"], state["skill_version_ref"])
    role_verdict = await judge_agent.judgmental_verdict(
        subject_id=target.skill_id, template_key="role_persona_conflict",   # 新增模板：静态审查
        content={"skill_md": target.body_markdown, "noise_pack_descriptions": [s.description for s in state["_noise_pack"]]},
        criticality=Criticality.ROUTINE,
    )

    temporal_findings = []
    for case in state["_multi_skill_cases"][: settings.multi_skill.max_temporal_probe_cases]:
        shuffled_case = case.model_copy(update={"prompt": _shuffle_step_order(case.prompt)})
        trace = await executor_backend.execute(ExecutionRequest(skill=target, case=shuffled_case, run_index=0, background_skills=state["_noise_pack"]))
        if not trace.loaded_skill_md or (trace.actions and trace.actions[-1].action_type == "internal_error"):
            temporal_findings.append(f"[拓扑脆弱] {case.case_id} 打乱步骤顺序后执行崩溃或未触发")

    return {**state, "judge_verdict_ids": [role_verdict.verdict_id], "_temporal_findings": temporal_findings}
```

`_shuffle_step_order()` 是简单的启发式文本重排（按句子/步骤分隔符打乱顺序），不追求语义完美，目的只是制造"逆序调用"的压力场景。

## 10. `core_skill_regression_gate`：基石技能增量回归熔断

```python
async def core_skill_regression_gate(state: PipelineState) -> PipelineState:
    target = await skill_repository.get(state["skill_id"], state["skill_version_ref"])
    breaches = []
    for core_skill in state["_core_skills"]:
        core_positive_cases = await test_case_repository.list_by_categories_for_skill(core_skill.skill_id, [TestCaseCategory.POSITIVE])
        sample = core_positive_cases[: settings.multi_skill.core_regression_sample_size]   # 默认5条,控制成本
        pass_count = 0
        for case in sample:
            trace = await executor_backend.execute(ExecutionRequest(skill=core_skill, case=case, run_index=0, background_skills=[target]))
            if trace.loaded_skill_md:
                pass_count += 1
        rate = pass_count / len(sample) if sample else 1.0
        if rate < settings.multi_skill.core_regression_min_rate:   # 默认0.8
            breaches.append(f"[基石熔断] 引入本Skill导致核心Skill '{core_skill.skill_id}' 触发率跌至{rate:.0%}")
    return {**state, "_core_regression_breaches": breaches}
```

**这一节点是架构文档"熔断拦截"机制的直接实现**：不是测目标 Skill 自己好不好，而是反过来测"引入这个新 Skill 是否伤害了系统里已有的、调用频率最高的核心 Skill"——因此 `background_skills=[target]`（把目标技能作为核心技能执行时的干扰背景），角色与其他节点相反。这是全模块十中**唯一默认阻断合并**的检测点（见第 11 节），因为它直接对应架构文档"流水线将直接阻断此次合并"的强约束。

## 11. 报告聚合与阻断策略

```python
async def finalize_dimension_report(state: PipelineState) -> PipelineState:
    hard_findings = state["_core_regression_breaches"]   # 唯一阻断项
    soft_findings = (
        state["_namespace_findings"] + state["_hijack_findings"] + state["_antagonism_findings"]
        + state["_attention_decay_findings"] + state["_temporal_findings"]
    )
    await report_generator.record_dimension_result(
        run_id=state["run_id"],
        result=DimensionResult(
            dimension="multi_skill_conflict",
            status=JudgeVerdictStatus.FAIL if (hard_findings or soft_findings) else JudgeVerdictStatus.PASS,
            score=None, findings=[*hard_findings, *soft_findings],
            blocking=bool(hard_findings),   # 只有基石回归熔断阻断；其余冲突现象作为高优先级告警，不阻断
        ),
    )
    if hard_findings or len(soft_findings) >= settings.multi_skill.deep_conflict_alert_threshold:   # 默认3
        await trigger_deep_conflict_alert(state["run_id"], hard_findings, soft_findings)   # 见第12节
    return state
```

**阻断策略取舍**：基石回归熔断（第 10 节）是唯一默认阻断项，与架构文档原文"流水线将直接阻断此次合并"精确对应；其余冲突现象（劫持、拮抗、注意力衰减、拓扑脆弱、命名污染）本文档定为不阻断但触发**深度冲突告警**——理由是这些现象大多需要人工判断"是背景干扰包的问题还是目标 Skill 的问题"（不像基石回归那样有清晰的因果方向：新 Skill 引入导致老 Skill 变差，责任明确），贸然自动阻断容易造成大量误伤,因此走告警+人工审查路径,延续本文档系列一贯的"确定性硬伤阻断、复杂/主观问题走人工"分层原则。
